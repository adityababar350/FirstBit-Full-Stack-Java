// 1. Write a program to print  sum of prime numbers in a given range.
#include <stdio.h>

int main()
{
    int start, end, i, j, sum = 0, flag;

    printf("Enter starting number: ");
    scanf("%d", &start);

    printf("Enter ending number: ");
    scanf("%d", &end);

    for(i = start; i <= end; i++)
    {
        if(i < 2)
            continue;

        flag = 1;

        for(j = 2; j < i; j++)
        {
            if(i % j == 0)
            {
                flag = 0;
                break;
            }
        }

        if(flag == 1)
        {
            printf("%d ", i);
            sum = sum + i;
        }
    }

    printf("\nSum of prime numbers = %d", sum);

    return 0;
}

//2. Write a program to accept string from user and replace one character from special symbol. 
#include <stdio.h>
#include <string.h>

int main()
{
    char str[100];
    char ch, symbol;
    int i;

    printf("Enter a string: ");
    gets(str);

    printf("Enter character to replace: ");
    scanf(" %c", &ch);

    printf("Enter special symbol: ");
    scanf(" %c", &symbol);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == ch)
        {
            str[i] = symbol;
        }
    }

    printf("Updated string = %s", str);

    return 0;
}

int addElement(int* arr, int* ci, int ele, int size) 
{
    if(*ci==size)  
    {
        printf("Array is full\n");
        return 0;
    }
    arr[*ci]=ele;
    (*ci)++;
}

void displayArray(int* arr, int ci) 
{
    if(ci==0)
    {
        printf("\nArray is empty");
        return;
    }
    
    for(int i=0; i<ci; i++)
        printf("%d ", arr[i]);
}

int searchElement(int* arr,int ci,int ele) 
{
	for(int i=0;i<ci;i++) 
	{
		if(arr[i]==ele) 
			return i;
	}
	return -1;
}

int deleteElement(int* arr,int* ci,int ele)
{
	int index=searchElement(arr,  *ci , ele);  
	if(index!=-1)   
	{
		for(int i=index;i<*ci-1;i++)  
			arr[i]=arr[i+1];
			
		(*ci)--;   
		return 1;  
	}
	else 
		return 0;
	
}

int updateElement(int* arr, int ci, int oldEle, int newEle) 
{
    int index=searchElement(arr,ci,oldEle);   

    if(index!=-1)
    {
        arr[index]=newEle;    
        return 1;   
    }

    return 0;   
}
